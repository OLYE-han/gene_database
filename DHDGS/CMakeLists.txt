cmake_minimum_required(VERSION 3.20)
project(DHDGS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
                OUTPUT_VARIABLE PYBIND_CMAKE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${PYBIND_CMAKE}")
find_package(pybind11 CONFIG REQUIRED)
set(RABITQ_BUILD_INDEX OFF CACHE BOOL "" FORCE)
set(RABITQ_BUILD_FASTSCAN OFF CACHE BOOL "" FORCE)
set(RABITQ_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(RABITQ_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RABITQ_USE_OPENMP OFF CACHE BOOL "" FORCE)
set(RABITQ_ENABLE_NATIVE_OPTIMIZATION OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../Reference_Code/RaBitQ-Library" rabitq)
pybind11_add_module(_native cpp/native.cpp cpp/async_io.cpp)
find_path(URING_INCLUDE_DIR liburing.h)
find_library(URING_LIBRARY uring)
if(URING_INCLUDE_DIR AND URING_LIBRARY)
  target_include_directories(_native PRIVATE ${URING_INCLUDE_DIR})
  target_link_libraries(_native PRIVATE ${URING_LIBRARY})
  target_compile_definitions(_native PRIVATE DHDGS_LIBURING=1)
endif()
target_link_libraries(_native PRIVATE rabitqlib::core)
target_compile_options(_native PRIVATE -Wall -Wextra)
set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/python/dhdgs")
