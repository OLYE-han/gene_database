                OUTPUT_VARIABLE PYBIND_CMAKE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)

find_path(URING_INCLUDE_DIR liburing.h)
find_library(URING_LIBRARY uring)
if(URING_INCLUDE_DIR AND URING_LIBRARY)
  target_include_directories(_native PRIVATE ${URING_INCLUDE_DIR})
  # Some hosts have a newer liburing.so development link but an older
  # liburing.so.2 in the loader's default search path. Inline header functions
  # then see a different ring layout (e.g. a zero SQ capacity after queue_init).
  # Stage the exact library selected by CMake under its SONAME, alongside this
  # module, instead of letting the loader substitute another system copy.
  get_filename_component(URING_REAL_LIBRARY "${URING_LIBRARY}" REALPATH)
  execute_process(COMMAND "${CMAKE_OBJDUMP}" -p "${URING_REAL_LIBRARY}"
                  OUTPUT_VARIABLE URING_ELF RESULT_VARIABLE URING_ELF_STATUS)
  string(REGEX MATCH "SONAME[ \t]+([^ \r\n]+)" URING_SONAME_MATCH "${URING_ELF}")
  set(URING_SONAME "${CMAKE_MATCH_1}")
  if(NOT URING_ELF_STATUS EQUAL 0 OR NOT URING_SONAME MATCHES "^liburing\\.so\\.[0-9]+$")
    message(FATAL_ERROR "Cannot identify the selected liburing shared-library SONAME")
  endif()
  set(URING_RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/python/dhdgs/.native_libs")
  file(MAKE_DIRECTORY "${URING_RUNTIME_DIR}")
  configure_file("${URING_REAL_LIBRARY}" "${URING_RUNTIME_DIR}/${URING_SONAME}" COPYONLY)
  target_link_libraries(_native PRIVATE "${URING_RUNTIME_DIR}/${URING_SONAME}")
  set_property(TARGET _native APPEND PROPERTY BUILD_RPATH "$ORIGIN/.native_libs")
  target_compile_definitions(_native PRIVATE DHDGS_LIBURING=1)
endif()
target_link_libraries(_native PRIVATE rabitqlib::core)

option(DHDGS_NATIVE_OPTIMIZATION "Optimize for the CPU of the build host" OFF)

  target_compile_options(_native PRIVATE -march=native -fopenmp-simd)

set(DHDGS_PYTHON_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/python/dhdgs" CACHE PATH "Native module staging directory")

option(DHDGS_ENABLE_RDMA "Enable optional rdma-core rsockets transport" ON)
if(DHDGS_ENABLE_RDMA)
  find_path(RDMACM_INCLUDE_DIR rdma/rsocket.h)
  find_library(RDMACM_LIBRARY rdmacm)
  if(RDMACM_INCLUDE_DIR AND RDMACM_LIBRARY)
    target_include_directories(_native PRIVATE ${RDMACM_INCLUDE_DIR})
    target_link_libraries(_native PRIVATE ${RDMACM_LIBRARY})
    target_compile_definitions(_native PRIVATE DHDGS_RDMA=1)
  else()
    message(STATUS "rdma-core not found: TCP-only build; selecting RDMA at runtime will fail")
  endif()
endif()
